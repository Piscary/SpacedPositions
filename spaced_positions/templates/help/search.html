rook ending{% translate "both words, in the title, the source, the tags or the comments" %} "back rank"{% translate "the words together, as written" %} #pin{% translate "positions with this tag" %}

{% translate "Terms: key:value" %}

motif:fork{% translate "a tag of the motifs, with all the tags under it (a knight fork is a fork)" %} motif:=fork{% translate "only this tag, not the ones under it" %} motif:fork,pin{% translate "either of the two (a comma means “or”)" %} src:"Endgame Manual"{% translate "one source; quotes keep a name with spaces together" %} has:solution{% translate "with a puzzle in training; also has:analysis, has:comment, has:puzzle, has:tags" %} no:solution{% translate "without a puzzle in training yet: the ones still to do" %} is:diagram{% translate "diagrams (positions that may be illegal and are not played); also is:draft, is:flagged, is:trash" %} side:w{% translate "White to move (b: Black)" %}

{% translate "Numbers and dates" %}

diff:3{% translate "difficulty 3 (of 1 to 5)" %} diff:>=3{% translate "3 or more; also >, <, <=" %} diff:2..4{% translate "from 2 to 4" %} page:12{% translate "page 12 of the source; ref: the diagram number" %} created:<7d{% translate "added in the last 7 days; also created:today, created:2026-10" %} pieces:<=7{% translate "at most 7 pieces on the board" %}

{% translate "Not" %}

-is:diagram{% translate "a minus in front leaves these out: everything that is not a diagram" %} not is:diagram{% translate "the same in words; !is:diagram and is:!=diagram work too" %} -motif:fork{% translate "no fork among the tags" %} -(src:a or src:b){% translate "a whole group left out" %}

{% translate "Or, and brackets" %}

motif:fork diff:>=3{% translate "terms side by side must all be true" %} motif:fork or motif:pin{% translate "or: one of them is enough" %} (src:a or src:b) diff:>=3{% translate "brackets group terms" %}

{% translate "Order" %}

sort:source{% translate "in the order of the book: source, page, diagram number, in groups per source" %} sort:-difficulty{% translate "hardest first (a minus reverses)" %} sort:created{% translate "oldest first; sort:-created newest first" %}

{% translate "Recipes" %}

{% key "library.analyse" as analyse %}{% key "analysis.subject" "next" as next %}{% key "library.train" as train %}
{% translate "Work through a book" %}
{% blocktranslate trimmed %}no:solution -is:diagram src:"…" sort:source, then {{ analyse }}: Analyse opens the first one. Once a puzzle is in training, it leaves the list and {{ next }} opens the next one still to do.{% endblocktranslate %}
{% translate "Train one motif" %}
{% blocktranslate trimmed %}motif:fork has:solution, then {{ train }}.{% endblocktranslate %}
{% translate "Find what is missing" %}
{% blocktranslate trimmed %}no:tags positions without tags, no:source without a source, is:draft puzzles not yet in training.{% endblocktranslate %}
{% translate "Keep a search" %}
{% blocktranslate trimmed %}:save-query name saves it; in:q:name uses it in another search and in custom training sessions.{% endblocktranslate %}
